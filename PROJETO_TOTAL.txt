/**
 * package-extension.js
 * Packages the extension into a distributable .zxp  
hive.
 *
 * Usage: node scripts/package-extension.js [--sign]
 * Requires: archiver (npm install archiver --save-dev)
 *
 * For production signing, use ZXPSignCmd from Adobe:
 *   https://github.com/Adobe-CEP/CEP-Resources/tree/master/ZXPSignCMD
 */

'use strict';

const path    = require('path');
const fs      = require('fs');
const archiver = require('archiver');

const ROOT    = path.resolve(__dirname, '..');
const DIST    = path.join(ROOT, 'dist');
const VERSION = require(path.join(ROOT, 'package.json')).version;
const ZXP     = path.join(DIST, 'DeepClean-' + VERSION + '.zxp');

// Files/dirs to include
const INCLUDE = [
  'CSXS',
  'jsx',
  'node',
  'panel',
  'package.json'
];

// Files/dirs to exclude
const EXCLUDE = [
  'node_modules',
  'dist',
  'scripts',
  '.git',
  '.babelrc',
  'jsx_compiled'
];

if (!fs.existsSync(DIST)) fs.mkdirSync(DIST, { recursive: true });

const output  = fs.createWriteStream(ZXP);
const archive = archiver('zip', { zlib: { level: 9 } });

output.on('close', () => {
  const kb = (archive.pointer() / 1024).toFixed(1);
  console.log('✓ Package created: ' + ZXP);
  console.log('  Size: ' + kb + ' KB');
  console.log('\nTo sign for distribution:');
  console.log('  ZXPSignCmd -sign . dist/DeepClean-' + VERSION + '.zxp cert.p12 <password>');
});

archive.on('error', err => { throw err; });
archive.pipe(output);

INCLUDE.forEach(item => {
  const fullPath = path.join(ROOT, item);
  if (!fs.existsSync(fullPath)) { console.warn('  Skip (not found): ' + item); return; }
  const stat = fs.statSync(fullPath);
  if (stat.isDirectory()) {
    archive.directory(fullPath, item, (data) => {
      // Exclude node_modules inside node/
      if (data.name.includes('node_modules')) return false;
      return data;
    });
  } else {
    archive.file(fullPath, { name: item });
  }
  console.log('  Added: ' + item);
});

archive.finalize();

atibility.
 * Run: node scripts/transpile-jsx.js
 */

'use strict';

const path  = require('path');
const fs    = require('fs');

let babel;
try {
  babel = require('@babel/core');
} catch (e) {
  console.warn('Babel not found — skipping JSX transpilation. Run npm install first.');
  process.exit(0);
}

const root    = path.resolve(__dirname, '..');
const jsxDir  = path.join(root, 'jsx');
const outDir  = path.join(root, 'jsx_compiled');

if (!fs.existsSync(outDir)) fs.mkdirSync(outDir, { recursive: true });

const files = fs.readdirSync(jsxDir).filter(f => f.endsWith('.jsx'));

files.forEach(file => {
  const src  = fs.readFileSync(path.join(jsxDir, file), 'utf8');
  let result;
  try {
    result = babel.transformSync(src, {
      filename: file,
      presets: [
        ['@babel/preset-env', {
          targets: { ie: '9' },
          modules: 'commonjs',
          useBuiltIns: false
        }]
      ]
    });
  } catch (err) {
    // Some ExtendScript constructs may fail Babel; fall back to raw source
    console.warn('Babel failed for ' + file + ': ' + err.message + ' — using raw source');
    result = { code: src };
  }
  const outPath = path.join(outDir, file);
  fs.writeFileSync(outPath, result.code, 'utf8');
  console.log('  Transpiled: ' + file + ' → jsx_compiled/' + file);
});

console.log('\nTranspilation complete. (' + files.length + ' files)');
console.log('Note: jsx/ originals are also used directly — transpiled copies are optional.');

/**
 * dev-install.js
 * Creates a symlink from the AE CEP extensions directory to this project,
 * enabling live development without re-packaging.
 *
 * Run: node scripts/dev-install.js
 * Requires: AE "PlayerDebugMode" registry key / plist set to 1.
 */

'use strict';

const fs     = require('fs');
const path   = require('path');
const os     = require('os');
const { execSync } = require('child_process');

const EXTENSION_ID   = 'com.deepclean.aep';
const PROJECT_ROOT   = path.resolve(__dirname, '..');

function getCepExtensionsDir() {
  const platform = os.platform();
  if (platform === 'win32') {
    const appData = process.env.APPDATA || path.join(os.homedir(), 'AppData', 'Roaming');
    return path.join(appData, 'Adobe', 'CEP', 'extensions');
  } else if (platform === 'darwin') {
    return path.join(os.homedir(), 'Library', 'Application Support', 'Adobe', 'CEP', 'extensions');
  }
  throw new Error('Unsupported platform: ' + platform);
}

function enableDebugMode() {
  const platform = os.platform();
  if (platform === 'darwin') {
    try {
      execSync('defaults write com.adobe.CSXS.11 PlayerDebugMode 1');
      execSync('defaults write com.adobe.CSXS.10 PlayerDebugMode 1');
      execSync('defaults write com.adobe.CSXS.9  PlayerDebugMode 1');
      console.log('✓ PlayerDebugMode enabled via defaults');
    } catch (e) {
      console.warn('Could not set PlayerDebugMode: ' + e.message);
    }
  } else if (platform === 'win32') {
    console.log('⚠ On Windows, manually set registry key:');
    console.log('  HKCU\\Software\\Adobe\\CSXS.11\\PlayerDebugMode = 1');
  }
}

try {
  const extDir = getCepExtensionsDir();
  const linkPath = path.join(extDir, EXTENSION_ID);

  // Create extensions dir if needed
  if (!fs.existsSync(extDir)) {
    fs.mkdirSync(extDir, { recursive: true });
    console.log('Created: ' + extDir);
  }

  // Remove existing
  if (fs.existsSync(linkPath)) {
    const stat = fs.lstatSync(linkPath);
    if (stat.isSymbolicLink()) {
      fs.unlinkSync(linkPath);
      console.log('Removed existing symlink: ' + linkPath);
    } else {
      console.error('ERROR: ' + linkPath + ' exists and is not a symlink. Remove manually.');
      process.exit(1);
    }
  }

  // Create symlink
  fs.symlinkSync(PROJECT_ROOT, linkPath, 'dir');
  console.log('✓ Symlink created:');
  console.log('  ' + linkPath + ' → ' + PROJECT_ROOT);

  enableDebugMode();

  console.log('\n✓ Dev install complete!');
  console.log('  1. Install node deps:  cd node && npm install');
  console.log('  2. Restart After Effects');
  console.log('  3. Window → Extensions → DeepClean');

} catch (err) {
  console.error('ERROR: ' + err.message);
  if (err.code === 'EPERM' || err.code === 'EACCES') {
    console.error('Try running with administrator / sudo privileges.');
  }
  process.exit(1);
}

    return String(str)
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }

  // ─── Transition to state ─────────────────────────────────────────────────────
  function transitionTo(newState) {
    state = newState;
    setButtonState();

    switch (newState) {
      case STATE.IDLE:
        showPanels(['idleState']);
        setStatus('Ready', '');
        activateStep(-1);
        break;
      case STATE.SNAPSHOT:
        showPanels(['progressPanel', 'logPanel']);
        setStatus('Running snapshot…', 'pulse');
        activateStep(0);
        setProgress(10, 'Collecting project data…');
        break;
      case STATE.ANALYZING:
        showPanels(['progressPanel', 'logPanel']);
        setStatus('Analysing dependencies…', 'pulse');
        activateStep(1);
        setProgress(40, 'Parsing .aepx + resolving graph…');
        break;
      case STATE.DRY_RUN:
        showPanels(['dryRunPanel', 'warningsPanel', 'logPanel']);
        setStatus('Ready to execute', 'ok');
        activateStep(2);
        setProgress(75, 'Analysis complete');
        break;
      case STATE.EXECUTING:
        showPanels(['progressPanel', 'logPanel']);
        setStatus('Executing…', 'pulse');
        activateStep(3);
        setProgress(80, 'Applying changes…');
        break;
      case STATE.DONE:
        showPanels(['logPanel']);
        setStatus('Done ✓', 'ok');
        activateStep(4);
        setProgress(100, 'Complete');
        showAlert('DeepClean complete! Use Ctrl+Z / Cmd+Z to undo.', 'success');
        break;
      case STATE.ERROR:
        showPanels(['logPanel']);
        setStatus('Error', 'error');
        break;
    }
  }

  // ─── evalScript wrapper (Promise) ───────────────────────────────────────────
  function evalScript(script) {
    return new Promise((resolve, reject) => {
      if (!csInterface) {
        // Dev/preview mode
        resolve('ERROR|CEP not available (preview mode)');
        return;
      }
      csInterface.evalScript(script, (result) => {
        if (result === 'EvalScript error.') {
          reject(new Error('evalScript failed: ' + script.substring(0, 80)));
        } else {
          resolve(result);
        }
      });
    });
  }

  // ─── Run Node.js analysis process ────────────────────────────────────────────
  function runAnalysis(snapshotJsonPath) {
    return new Promise((resolve, reject) => {
      if (!nodeCP) {
        reject(new Error('Node.js child_process not available'));
        return;
      }

      // Resolve path to analyze.js relative to this panel
      const extensionRoot = csInterface
        ? csInterface.getSystemPath(SystemPath.EXTENSION)
        : __dirname + '/..';

      const scriptPath = nodePath.join(extensionRoot, 'node', 'analyze.js');
      appendLog('Spawning analysis: ' + scriptPath, 'info');

      const child = nodeCP.spawn(process.execPath, [scriptPath, snapshotJsonPath], {
        cwd: nodePath.join(extensionRoot, 'node')
      });

      let stdout = '';
      let stderr = '';

      child.stdout.on('data', d => {
        const str = d.toString();
        stdout += str;
        str.split('\n').forEach(line => {
          if (line.trim()) appendLog(line, 'info');
        });
      });

      child.stderr.on('data', d => {
        const str = d.toString();
        stderr += str;
        str.split('\n').forEach(line => {
          if (line.trim()) appendLog(line, 'error');
        });
      });

      child.on('close', code => {
        if (code !== 0) {
          reject(new Error('Analysis process failed (exit ' + code + '): ' + stderr));
          return;
        }
        // Extract manifest path from last MANIFEST_READY| line
        const lines = stdout.split('\n');
        let mPath = null;
        for (let i = lines.length - 1; i >= 0; i--) {
          if (lines[i].startsWith('MANIFEST_READY|')) {
            mPath = lines[i].split('|')[1].trim();
            break;
          }
        }
        if (!mPath) {
          reject(new Error('Analysis did not output MANIFEST_READY'));
          return;
        }
        resolve(mPath);
      });

      child.on('error', err => {
        reject(new Error('Could not spawn analysis process: ' + err.message));
      });
    });
  }

  // ─── Read JSON file ───────────────────────────────────────────────────────────
  function readJsonFile(filePath) {
    if (!nodeFS) throw new Error('fs not available');
    const raw = nodeFS.readFileSync(filePath, 'utf8');
    return JSON.parse(raw);
  }

  // ─── FLOW: Analyse ────────────────────────────────────────────────────────────
  async function flowAnalyse() {
    clearAlert();
    dom.logOutput.innerHTML = '';

    try {
      // ── Step 1: Snapshot ─────────────────────────────────────────────────────
      transitionTo(STATE.SNAPSHOT);
      appendLog('Starting snapshot collection…', 'info');

      const snapResult = await evalScript('DC_runSnapshot()');
      appendLog('Snapshot result: ' + snapResult.substring(0, 120), 'info');

      if (!snapResult.startsWith('SUCCESS|')) {
        throw new Error(snapResult.replace('ERROR|', ''));
      }

      snapshotPath = snapResult.split('|')[1].trim();
      appendLog('Snapshot saved: ' + snapshotPath, 'ok');
      setProgress(30, 'Snapshot collected, starting analysis…');

      // ── Step 2: Analysis ─────────────────────────────────────────────────────
      transitionTo(STATE.ANALYZING);
      appendLog('Starting dependency analysis…', 'info');

      manifestPath = await runAnalysis(snapshotPath);
      appendLog('Manifest written: ' + manifestPath, 'ok');
      setProgress(70, 'Analysis complete, loading manifest…');

      // ── Read manifest ────────────────────────────────────────────────────────
      manifest = readJsonFile(manifestPath);
      appendLog('Manifest loaded — ' +
        manifest.stats.layersToRemove + ' layers to remove, ' +
        manifest.stats.itemsToLimbo + ' items to limbo', 'ok');

      // ── Show dry-run ─────────────────────────────────────────────────────────
      populateDryRun(manifest);
      transitionTo(STATE.DRY_RUN);

    } catch (err) {
      appendLog('ERROR: ' + err.message, 'error');
      showAlert('Analysis failed: ' + err.message, 'error');
      transitionTo(STATE.ERROR);
    }
  }

  // ─── FLOW: Execute ────────────────────────────────────────────────────────────
  async function flowExecute() {
    if (!manifestPath) {
      showAlert('No manifest available. Run analysis first.', 'error');
      return;
    }
    clearAlert();

    try {
      transitionTo(STATE.EXECUTING);
      appendLog('Executing manifest: ' + manifestPath, 'info');

      const execResult = await evalScript('DC_runExecution("' +
        manifestPath.replace(/\\/g, '\\\\') + '")');

      if (execResult.startsWith('SUCCESS|')) {
        const logLines = execResult.replace('SUCCESS|', '').split('\n');
        logLines.forEach(line => {
          const type = line.startsWith('[WARN') ? 'warn'
                     : line.startsWith('[DONE') ? 'ok' : 'info';
          if (line.trim()) appendLog(line, type);
        });
        transitionTo(STATE.DONE);
      } else {
        throw new Error(execResult.replace('ERROR|', ''));
      }
    } catch (err) {
      appendLog('EXECUTION ERROR: ' + err.message, 'error');
      showAlert('Execution failed: ' + err.message, 'error');
      transitionTo(STATE.ERROR);
    }
  }

  // ─── FLOW: Reset ─────────────────────────────────────────────────────────────
  function flowReset() {
    manifest     = null;
    snapshotPath = null;
    manifestPath = null;
    clearAlert();
    dom.logOutput.innerHTML = '';
    transitionTo(STATE.IDLE);
    pollProjectInfo();
  }

  // ─── Poll project info ────────────────────────────────────────────────────────
  async function pollProjectInfo() {
    try {
      const result = await evalScript('DC_getProjectInfo()');
      if (result && result.startsWith('SUCCESS|')) {
        const info = JSON.parse(result.split('|')[1]);
        projectInfo = info;
        updateProjectInfo(info);
      }
    } catch (e) {
      // Not in AE or CEP not ready
    }
  }

  // ─── Init ─────────────────────────────────────────────────────────────────────
  function init() {
    // Button bindings
    dom.btnAnalyze.addEventListener('click', () => {
      if (state === STATE.IDLE || state === STATE.ERROR || state === STATE.DONE) {
        flowAnalyse();
      }
    });

    dom.btnExecute.addEventListener('click', () => {
      if (state === STATE.DRY_RUN) flowExecute();
    });

    dom.btnReset.addEventListener('click', flowReset);

    // Initial state
    transitionTo(STATE.IDLE);

    // Poll project info every 3s
    pollProjectInfo();
    setInterval(pollProjectInfo, 3000);

    // CEP theme sync
    if (csInterface) {
      const skinInfo = csInterface.getHostEnvironment().appSkinInfo;
      if (skinInfo) {
        // Already dark-themed; AE skin sync optional
      }
    }
  }

  // Wait for DOM
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }

})();

/** Loads binary file created which is located at url asynchronously
*
*@param urlName url at which binary file is located. Local files should start with 'file://'
*@param callback Optional. A callback function that returns after binary is loaded

*@example
* To create JS binary use command ./cep_compiler test.js test.bin
* To load JS binary asyncronously
*   var CSLib = new CSInterface();
*   CSLib.loadBinAsync(url, function () { });
*/
CSInterface.prototype.loadBinAsync = function(urlName,callback)
{
    try
    {
        var xhr = new XMLHttpRequest();
        xhr.responseType = 'arraybuffer'; // make response as ArrayBuffer
        xhr.open('GET', urlName, true);
        xhr.onerror = function ()
        {
  		  console.log("Unable to load snapshot from given URL");
  		  return false;
		};
        xhr.send();
        xhr.onload = () => {
            window.__adobe_cep__.loadSnapshot(xhr.response);
            if (typeof callback === "function")
            {
                callback();
            }
            else if(typeof callback !== "undefined")
            {
                console.log("Provided callback is not a function");
            }
        }
    }
    catch(err)
    {
        console.log(err);
        return false;
    }

	return true;
};

/** Loads binary file created synchronously
*
*@param pathName the local path at which binary file is located

*@example
* To create JS binary use command ./cep_compiler test.js test.bin
* To load JS binary syncronously
*   var CSLib = new CSInterface();
*   CSLib.loadBinSync(path);
*/
CSInterface.prototype.loadBinSync  = function(pathName)
{
    try
    {
        var OSVersion = this.getOSInformation();
        if(pathName.startsWith("file://"))
        {
            if (OSVersion.indexOf("Windows") >= 0)
            {
               pathName = pathName.replace("file:///", "");
            }
            else if (OSVersion.indexOf("Mac") >= 0)
            {
                pathName = pathName.replace("file://", "");
            }
            window.__adobe_cep__.loadSnapshot(pathName);
            return true;
        }
    }
    catch(err)
    {
        console.log(err);
        return false;
    }
    //control should not come here
    return false;
};

/**
 * Initializes the resource bundle for this extension with property values
 * for the current application and locale.
 * To support multiple locales, you must define a property file for each locale,
 * containing keyed display-string values for that locale.
 * See localization documentation for Extension Builder and related products.
 *
 * Keys can be in the
 * form <code>key.value="localized string"</code>, for use in HTML text elements.
 * For example, in this input element, the localized \c key.value string is displayed
 * instead of the empty \c value string:
 *
 * <code><input type="submit" value="" data-locale="key"/></code>
 *
 * @return An object containing the resource bundle information.
 */
CSInterface.prototype.initResourceBundle = function()
{
    var resourceBundle = JSON.parse(window.__adobe_cep__.initResourceBundle());
    var resElms = document.querySelectorAll('[data-locale]');
    for (var n = 0; n < resElms.length; n++)
    {
       var resEl = resElms[n];
       // Get the resource key from the element.
       var resKey = resEl.getAttribute('data-locale');
       if (resKey)
       {
           // Get all the resources that start with the key.
           for (var key in resourceBundle)
           {
               if (key.indexOf(resKey) === 0)
               {
                   var resValue = resourceBundle[key];
                   if (key.length == resKey.length)
                   {
                        resEl.innerHTML = resValue;
                   }
                   else if ('.' == key.charAt(resKey.length))
                   {
                        var attrKey = key.substring(resKey.length + 1);
                        resEl[attrKey] = resValue;
                   }
               }
           }
       }
    }
    return resourceBundle;
};

/**

 */

{
    var userAgent = navigator.userAgent;

    if ((navigator.platform == "Win32") || (navigator.platform == "Windows"))
    {
        var winVersion = "Windows";
        var winBit = "";
        if (userAgent.indexOf("Windows") > -1)
        {
            if (userAgent.indexOf("Windows NT 5.0") > -1)
            {
                winVersion = "Windows 2000";
            }
            else if (userAgent.indexOf("Windows NT 5.1") > -1)
            {
                winVersion = "Windows XP";
            }
            else if (userAgent.indexOf("Windows NT 5.2") > -1)
            {
                winVersion = "Windows Server 2003";
            }
            else if (userAgent.indexOf("Windows NT 6.0") > -1)
            {
                winVersion = "Windows Vista";
            }
            else if (userAgent.indexOf("Windows NT 6.1") > -1)
            {
                winVersion = "Windows 7";
            }
            else if (userAgent.indexOf("Windows NT 6.2") > -1)
            {
                winVersion = "Windows 8";
            }

// ─── STEP 1: Clean timelines ──────────────────────────────────────────────────
function cleanTimelines(proj, manifest, log) {
  log.push('[STEP 1] Cleaning timelines...');

  var layersToRemove = manifest.layersToRemove || [];
  if (layersToRemove.length === 0) {
    log.push('[STEP 1] No layers to remove.');
    return;
  }

  // Group by compId
  var removeMap = {};
  for (var j = 0; j < layersToRemove.length; j++) {
    var rem = layersToRemove[j];
    var cid = String(rem.compId);
    if (!removeMap[cid]) removeMap[cid] = [];
    removeMap[cid].push(rem.layerIndex);
  }

  for (var compIdStr in removeMap) {
    var comp = findCompById(proj, parseInt(compIdStr, 10));
    if (!comp) {
      log.push('[WARN] Comp id ' + compIdStr + ' not found — skipping.');
      continue;
    }

    // Sort DESCENDING so removing higher indices first doesn't shift lower ones
    var indices = removeMap[compIdStr].slice().sort(function (a, b) { return b - a; });

    // Build whitelist set for this comp (indices to KEEP)
    var whitelist = manifest.whitelist || [];
    var keepSet = {};
    for (var wi = 0; wi < whitelist.length; wi++) {
      if (String(whitelist[wi].compId) === compIdStr) {
        keepSet[String(whitelist[wi].layerIndex)] = true;
      }
    }

    var removed = 0;
    var skipped = 0;
    for (var k = 0; k < indices.length; k++) {
      var idx = indices[k];
      // Safety: never remove if it appears in the whitelist
      if (keepSet[String(idx)]) { skipped++; continue; }
      try {
        if (idx >= 1 && idx <= comp.numLayers) {
          comp.layer(idx).remove();
          removed++;
        }
      } catch (e) {
        log.push('[WARN] Could not remove layer ' + idx + ' from "' + comp.name + '": ' + e.message);
      }
    }
    log.push('[STEP 1] "' + comp.name + '" — removed ' + removed + ' layers' +
             (skipped > 0 ? ', skipped ' + skipped + ' (whitelisted)' : ''));
  }
}

// ─── Safe JSON stringify (ES3 fallback) ───────────────────────────────────────
function safeStringify(obj) {
  if (typeof JSON !== 'undefined' && JSON.stringify) {
    return JSON.stringify(obj, null, 2);
  }
  return _stringify(obj);
}
function _stringify(val) {
  var type = typeof val;
  if (val === null) return 'null';
  if (type === 'boolean') return val ? 'true' : 'false';
  if (type === 'number') return isFinite(val) ? String(val) : 'null';
  if (type === 'string') {
    return '"' + val.replace(/\\/g, '\\\\').replace(/"/g, '\\"')
      .replace(/\n/g, '\\n').replace(/\r/g, '\\r').replace(/\t/g, '\\t') + '"';
  }
  if (val instanceof Array) {
    var arrParts = [];
    for (var i = 0; i < val.length; i++) arrParts.push(_stringify(val[i]));
    return '[' + arrParts.join(',') + ']';
  }
  if (type === 'object') {
    var objParts = [];
    for (var k in val) {
      if (val.hasOwnProperty(k)) {
        objParts.push(_stringify(k) + ':' + _stringify(val[k]));
      }
    }
    return '{' + objParts.join(',') + '}';
  }
  return 'null';
}

// ─── Stale check: wait until file size stabilises ────────────────────────────
function waitUntilStable(filePath, maxWaitMs, intervalMs) {
  maxWaitMs = maxWaitMs || 15000;
  intervalMs = intervalMs || 400;
  var f = new File(filePath);
  var lastSize = -1;
  var elapsed = 0;
  while (elapsed < maxWaitMs) {
    $.sleep(intervalMs);
    elapsed += intervalMs;
    f.open('r');
    var size = f.length;
    f.close();
    if (size > 0 && size === lastSize) return true;
    lastSize = size;
  }
  return false;
}

// ─── Hex → UTF-8 (for <bdata> plugin blobs) ──────────────────────────────────
function hexToUtf8(hex) {
  try {
    const buf = Buffer.from(hex.replace(/\s/g, ''), 'hex');
    return buf.toString('utf8');
  } catch (e) { return ''; }
}

// ─── Safe acorn parse ─────────────────────────────────────────────────────────
function safeAcornParse(src) {
  try {
    return acorn.parse(src, { ecmaVersion: 2020, sourceType: 'script' });
  } catch (e) {
    try {
      return acorn.parse(src, { ecmaVersion: 2020, sourceType: 'module' });
    } catch (e2) {
      return null;
    }
  }
}

// ─── Walk AST and collect string literals / identifiers ─────────────────────
function walkAst(node, collector, depth) {
  if (!node || typeof node !== 'object' || depth > 50) return;
  if (node.type === 'Literal' && typeof node.value === 'string') {
    collector.push(node.value);
  }
  if (node.type === 'Identifier') {
    collector.push(node.name);
  }
  for (const key of Object.keys(node)) {
    if (key === 'type' || key === 'start' || key === 'end') continue;
    const child = node[key];
    if (Array.isArray(child)) {
      child.forEach(c => walkAst(c, collector, depth + 1));
    } else if (child && typeof child === 'object') {
      walkAst(child, collector, depth + 1);
    }
  }
}

// ─── Detect dynamic expressions ──────────────────────────────────────────────
// Dynamic expressions use eval(), Function(), computed property access, or
// network/file I/O — we cannot statically resolve these.
function isDynamic(exprSrc) {
  const dynamicPatterns = [
    /\beval\s*\(/,
    /\bnew\s+Function\s*\(/,
    /\[.*\]/,                  // computed bracket access
    /\bthisComp\.layer\s*\([^"']/,  // dynamic layer lookup
    /\bcomp\s*\([^"']/,             // dynamic comp lookup
    /\.sourceText\s*\.\s*value/,
    /\bfootage\s*\([^"']/,
  ];
  return dynamicPatterns.some(p => p.test(exprSrc));
}

// ─── Extract comp/item names from expression via AST ─────────────────────────
function extractExpressionRefs(exprSrc) {
  const refs = { strings: [], isDynamic: isDynamic(exprSrc) };
  const ast = safeAcornParse(exprSrc);
  if (!ast) {
    // Fallback: naive string extraction (for legacy ES3 expressions)
    const strMatches = exprSrc.match(/["'`]([^"'`]{2,64})["'`]/g) || [];
    strMatches.forEach(m => refs.strings.push(m.replace(/["'`]/g, '')));
    return refs;
  }
  walkAst(ast, refs.strings, 0);
  return refs;
}

// ─── Parse .aepx via streaming SAX parser ────────────────────────────────────
function parseAepx(aepxPath) {
  return new Promise((resolve, reject) => {
    if (!aepxPath || !fs.existsSync(aepxPath)) {
      log('No .aepx found, skipping XML analysis');
      return resolve({ layerSources: {}, pluginRefs: [], bdataItems: [] });
    }

    log('Parsing .aepx: ' + aepxPath);

    const result = {
      layerSources: {},   // { compName: [sourceName, ...] }
      pluginRefs: [],     // [{ compName, layerName, pluginData }]
      bdataItems: []      // raw bdata strings for further analysis
    };

    const parser = sax.createStream(false, { lowercase: true });

    let currentComp  = null;
    let currentLayer = null;
    let insideBdata  = false;
    let bdataBuffer  = '';

    parser.on('opentag', (node) => {
      const name = node.name;
      if (name === 'compitem' || name === 'comp') {
        currentComp = node.attributes.name || node.attributes['ae:name'] || null;
      }
      if (name === 'layer' || name === 'avlayer') {
        currentLayer = node.attributes.name || node.attributes['ae:name'] || null;
      }
      if (name === 'bdata') {
        insideBdata = true;
        bdataBuffer = '';
      }
    });

    parser.on('text', (text) => {
      if (insideBdata) bdataBuffer += text;
    });

    parser.on('closetag', (name) => {
      if (name === 'bdata') {
        insideBdata = false;
        if (bdataBuffer.trim().length > 0) {
          const decoded = hexToUtf8(bdataBuffer.trim());
          if (decoded.length > 3) {
            result.bdataItems.push(decoded);
            if (currentComp && currentLayer) {
              result.pluginRefs.push({
                compName:   currentComp,
                layerName:  currentLayer,
                pluginData: decoded
              });
            }
          }
        }
        bdataBuffer = '';
      }
      if (name === 'compitem' || name === 'comp') currentComp  = null;
      if (name === 'layer'    || name === 'avlayer') currentLayer = null;
    });

    parser.on('error', (e) => {
      warn('SAX parser error (non-fatal): ' + e.message);
      parser._parser.error = null;
      parser._parser.resume();
    });

    parser.on('end', () => {
      log('AEPX parse complete. bdata entries: ' + result.bdataItems.length);
      resolve(result);
    });

    try {
      fs.createReadStream(aepxPath, { encoding: 'utf8' }).pipe(parser);
    } catch (e) {
      warn('Could not stream .aepx: ' + e.message);
      resolve(result);
    }
  });
}

// ─── Build item lookup maps ───────────────────────────────────────────────────
function buildItemMaps(allItems) {
  const byId   = {};
  const byName = {};
  for (const item of allItems) {
    byId[item.id]    = item;
    if (!byName[item.name]) byName[item.name] = [];
    byName[item.name].push(item);
  }
  return { byId, byName };
}

// ─── Dependency graph node ────────────────────────────────────────────────────
class DepGraph {
  constructor() {
    this.nodes = new Set();   // item ids
    this.edges = {};          // id → Set<id>
  }
  addNode(id) {
    this.nodes.add(id);
    if (!this.edges[id]) this.edges[id] = new Set();
  }
  addEdge(fromId, toId) {
    this.addNode(fromId);
    this.addNode(toId);
    this.edges[fromId].add(toId);
  }
  // Resolve all reachable nodes from seeds (BFS)
  resolve(seedIds) {
    const visited = new Set();
    const queue   = [...seedIds];
    while (queue.length) {
      const id = queue.shift();
      if (visited.has(id)) continue;
      visited.add(id);
      const deps = this.edges[id];
      if (deps) deps.forEach(d => queue.push(d));
    }
    return visited;
  }
}

// ─── Resolve expression dependencies ─────────────────────────────────────────
function resolveExpressionDeps(expressions, itemMaps, warnings) {
  const deps = new Set();
  for (const exprEntry of expressions) {
    const src = exprEntry.expression;
    const refs = extractExpressionRefs(src);
    if (refs.isDynamic) {
      warnings.push(
        'Dynamic expression found in property "' + exprEntry.propPath +
        '" — treating as UNUSED (aggressive mode). Expression: ' +
        src.substring(0, 80).replace(/\n/g, ' ') + '...'
      );
      continue; // Aggressive mode: skip dynamic
    }
    // Resolve string literals against item names
    for (const str of refs.strings) {
      if (itemMaps.byName[str]) {
        itemMaps.byName[str].forEach(item => deps.add(item.id));
      }
    }
  }
  return deps;
}

// ─── Resolve bdata plugin refs ────────────────────────────────────────────────
function resolvePluginDeps(bdataItems, itemMaps) {
  const deps = new Set();
  for (const raw of bdataItems) {
    // Extract anything that looks like a comp/footage name from plugin data
    // Strategy: scan for sequences that match item names
    for (const name of Object.keys(itemMaps.byName)) {
      if (name.length > 2 && raw.includes(name)) {
        itemMaps.byName[name].forEach(item => deps.add(item.id));
      }
    }
  }
  return deps;
}

  // Validate paths
  validatePaths(allItems, warnings);

  // ── 1. Seed selected comps into graph ───────────────────────────────────────
  const selectedIds = new Set(selectedComps.map(c => c.id));
  selectedIds.forEach(id => graph.addNode(id));

  // ── 2. Process layers in selected comps ─────────────────────────────────────
  const whitelist      = [];   // { compId, compName, layerIndex }
  const layersToRemove = [];   // { compId, compName, layerIndex }

  // Track which external comps are needed and which layers within them
  const externalCompMap = {};  // compId → Set<layerIndex>

  for (const selComp of selectedComps) {
    log('Analysing comp: ' + selComp.name + ' (' + selComp.layers.length + ' layers)');
    const compItem = itemMaps.byId[selComp.id];

    for (const layer of selComp.layers) {
      let keep = false;
      let reason = '';

      // Layer source → add to graph
      if (layer.sourceId) {
        const sourceItem = itemMaps.byId[layer.sourceId];
        if (sourceItem) {
          graph.addEdge(selComp.id, layer.sourceId);

          if (sourceItem.type === 'CompItem' && !selectedIds.has(layer.sourceId)) {
            // External comp dependency
            if (!externalCompMap[layer.sourceId]) {
              externalCompMap[layer.sourceId] = new Set();
            }
            // This layer is a "seed" in the external comp
            externalCompMap[layer.sourceId].add(layer.index);
          }
          keep = true;
          reason = 'has source';
        }
      }

      // Expression dependencies
      if (layer.expressions && layer.expressions.length > 0) {
        const exprDeps = resolveExpressionDeps(layer.expressions, itemMaps, warnings);
        if (exprDeps.size > 0) {
          exprDeps.forEach(depId => graph.addEdge(selComp.id, depId));
          keep = true;
          reason += ' +expressions';
        }
      }

      // Effect names — keep if any effects present
      if (layer.effectNames && layer.effectNames.length > 0) {
        keep = true;
        reason += ' +effects';
      }

      if (keep) {
        whitelist.push({ compId: selComp.id, compName: selComp.name, layerIndex: layer.index, reason });
      } else {
        layersToRemove.push({ compId: selComp.id, compName: selComp.name, layerIndex: layer.index });
      }
    }
  }

  // ── 3. Add plugin deps from bdata ────────────────────────────────────────────
  const pluginDeps = resolvePluginDeps(aepxData.bdataItems, itemMaps);
  pluginDeps.forEach(id => {
    graph.addNode(id);
    selectedIds.forEach(sId => graph.addEdge(sId, id));
  });

  // ── 4. Resolve full dependency graph from seeds ──────────────────────────────
  const reachableIds = graph.resolve([...selectedIds]);
  log('Reachable items: ' + reachableIds.size + ' / ' + allItems.length);

  // ── 5. Process external comps: selective pruning ─────────────────────────────
  const externalCompsManifest = [];

  // Build a lookup: compId (number) → layer data, from snapshot's extCompLayers
  // (populated from snapshot if deep mode was enabled, otherwise empty)
  const extCompLayerData = {};
  if (snapshot.externalCompLayers) {
    for (const entry of snapshot.externalCompLayers) {
      extCompLayerData[entry.compId] = entry.layers;
    }
  }

  for (const extCompIdStr of Object.keys(externalCompMap)) {
    const extCompId   = parseInt(extCompIdStr, 10);  // always number
    const extCompItem = itemMaps.byId[extCompId];
    if (!extCompItem) {
      warn('External comp id ' + extCompIdStr + ' not found in item map');
      continue;
    }

    // Circular dependency guard: if this comp is also a selected comp, skip nesting
    if (selectedIds.has(extCompId)) {
      warnings.push(
        'Circular dependency: comp "' + extCompItem.name +
        '" is both selected and referenced as external. Skipping DC_ANCHOR injection.'
      );
      continue;
    }

    const seedLayerSet = externalCompMap[extCompIdStr]; // Set<layerIndex>
    let   requiredLayers = [...seedLayerSet];

    // If we have deep layer data for this external comp, trace expression deps within it
    if (extCompLayerData[extCompId]) {
      const extLayers = extCompLayerData[extCompId];
      // BFS within external comp: seed → expression refs → sources
      const reqSet  = new Set(requiredLayers);
      const queue   = [...requiredLayers];
      while (queue.length) {
        const idx = queue.shift();
        const layer = extLayers.find(l => l.index === idx);
        if (!layer) continue;
        // Add source
        if (layer.sourceId && !reachableIds.has(layer.sourceId)) {
          reachableIds.add(layer.sourceId);
        }
        // Add expression string refs within the same comp
        if (layer.expressions) {
          const exprDeps = resolveExpressionDeps(layer.expressions, itemMaps, warnings);
          exprDeps.forEach(depId => reachableIds.add(depId));
        }
        // Include parent chain (layers with parenting)
        if (layer.parentIndex && !reqSet.has(layer.parentIndex)) {
          reqSet.add(layer.parentIndex);
          queue.push(layer.parentIndex);
        }
      }
      requiredLayers = [...reqSet];
    }

    // Mark external comp itself as reachable
    reachableIds.add(extCompId);

    externalCompsManifest.push({
      compId:         extCompId,
      compName:       extCompItem.name,
      requiredLayers: requiredLayers
    });

    log('External comp "' + extCompItem.name + '": keeping layers [' + requiredLayers.join(', ') + ']');
  }

  // ── 6. Determine items to move to Limbo ──────────────────────────────────────
  // [KEEP] override: any item whose name starts with "[KEEP]" is force-whitelisted
  const forceKeepIds = new Set();
  for (const item of allItems) {
    if (item.name && item.name.toUpperCase().startsWith('[KEEP]')) {
      forceKeepIds.add(item.id);
      log('[KEEP] override applied: "' + item.name + '"');
    }
  }

  const itemsToMoveToLimbo = [];

  // Identify "owner comp" for each item (which selected comp first uses it)
  const ownerMap = {};  // itemId → { compId, compName }
  for (const entry of whitelist) {
    const selComp = selectedComps.find(c => c.id === entry.compId);
    if (!selComp) continue;
    const layer = selComp.layers.find(l => l.index === entry.layerIndex);
    if (!layer) continue;
    // Register source as owned by this comp
    if (layer.sourceId && !ownerMap[layer.sourceId]) {
      ownerMap[layer.sourceId] = { compId: entry.compId, compName: entry.compName };
    }
  }
  // Also register items reached via expression deps as owned by the comp that references them
  for (const selComp of selectedComps) {
    for (const layer of selComp.layers) {
      if (!layer.expressions) continue;
      const exprDeps = resolveExpressionDeps(layer.expressions, itemMaps, []);
      exprDeps.forEach(depId => {
        if (!ownerMap[depId]) {
          ownerMap[depId] = { compId: selComp.id, compName: selComp.name };
        }
      });
    }
  }

  for (const item of allItems) {
    if (item.type === 'FolderItem') continue;
    if (reachableIds.has(item.id)) continue;
    if (forceKeepIds.has(item.id)) continue; // [KEEP] override

    // Skip the limbo folder itself if it somehow appears
    if (item.name === '_LIMBO_DeepClean') continue;

    const owner = ownerMap[item.id];
    itemsToMoveToLimbo.push({
      id:            item.id,
      name:          item.name,
      type:          item.type,
      ownerCompId:   owner ? owner.compId   : null,
      ownerCompName: owner ? owner.compName : null
    });
  }

  log('Items to move to limbo: ' + itemsToMoveToLimbo.length);

  // ── 7. Estimated size reduction ───────────────────────────────────────────────
  let estimatedSavedBytes = 0;
  for (const entry of itemsToMoveToLimbo) {
    if (entry.filePath) {
      try {
        const stat = fs.statSync(entry.filePath);
        estimatedSavedBytes += stat.size;
      } catch (e) {}
    }
  }

  // ── 8. Write manifest_execution.json ─────────────────────────────────────────
  const manifest = {
    version:              '1.0.0',
    timestamp:            new Date().toISOString(),
    projectName:          snapshot.projectName,
    whitelist,
    layersToRemove,
    itemsToMoveToLimbo,
    externalComps:        externalCompsManifest,
    warnings,
    stats: {
      selectedComps:      selectedComps.length,
      totalItems:         allItems.length,
      reachableItems:     reachableIds.size,
      itemsToLimbo:       itemsToMoveToLimbo.length,
      layersToRemove:     layersToRemove.length,
      warningCount:       warnings.length,
      estimatedSavedMB:   (estimatedSavedBytes / (1024 * 1024)).toFixed(2)
    }
  };

  const manifestPath = path.join(tempDir, 'manifest_execution.json');
  await fs.writeJson(manifestPath, manifest, { spaces: 2 });

  log('Manifest written: ' + manifestPath);
  // Final output consumed by panel
  process.stdout.write('MANIFEST_READY|' + manifestPath + '\n');
}

main().catch(err => {
  process.stderr.write('FATAL: ' + err.message + '\n' + err.stack + '\n');
  process.exit(1);
});
